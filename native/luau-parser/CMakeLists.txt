cmake_minimum_required(VERSION 3.31)
project(RoGraphLuau LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(LUAU_BUILD_CLI OFF CACHE BOOL "" FORCE)
set(LUAU_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LUAU_STATIC_CRT ON CACHE BOOL "" FORCE)
add_subdirectory(../../vendor/luau luau EXCLUDE_FROM_ALL)
add_executable(luau-parser main.cpp ../../vendor/luau/Analysis/src/AstJsonEncoder.cpp)
target_include_directories(luau-parser PRIVATE ../../vendor ../../vendor/luau/Analysis/include)
target_link_libraries(luau-parser PRIVATE Luau.Ast Luau.Common)
target_compile_features(luau-parser PRIVATE cxx_std_17)
